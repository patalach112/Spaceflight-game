cmake_minimum_required(VERSION 3.24)
project(AegisSpaceflight VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(AEGIS_BUILD_GAME "Build the Raylib application" ON)
option(AEGIS_BUILD_TESTS "Build the headless construction tests" ON)

add_library(aegis_build src/PartCatalog.cpp src/VABSystem.cpp)
target_include_directories(aegis_build PUBLIC include)
if(MSVC)
  target_compile_options(aegis_build PRIVATE /W4 /permissive-)
else()
  target_compile_options(aegis_build PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(AEGIS_BUILD_GAME)
  find_package(raylib 5.5 QUIET)
  if(NOT raylib_FOUND)
    include(FetchContent)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(raylib URL https://codeload.github.com/raysan5/raylib/tar.gz/refs/tags/5.5
      URL_HASH SHA256=aea98ecf5bc5c5e0b789a76de0083a21a70457050ea4cc2aec7566935f5e258e
      DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(raylib)
  endif()
  add_executable(aegis src/main.cpp src/Game.cpp src/Renderer.cpp src/UIComponents.cpp)
  target_link_libraries(aegis PRIVATE aegis_build raylib)
  if(MSVC)
    target_compile_options(aegis PRIVATE /W4 /permissive-)
  else()
    target_compile_options(aegis PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()

if(AEGIS_BUILD_TESTS)
  enable_testing()
  add_executable(aegis_tests tests/VABSystemTests.cpp)
  target_link_libraries(aegis_tests PRIVATE aegis_build)
  add_test(NAME construction COMMAND aegis_tests)
endif()
